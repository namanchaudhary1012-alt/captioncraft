const API_KEY = 'sk-or-v1-
787f3d9143250f478bc60adfdbee44ea0e4a0c27b2ec09df391661036a4107f1';
async function generate() {
const desc = document.getElementById('postDesc').value.trim();
if (!desc) { showError('Please describe your post first!'); return; }
const tone = document.getElementById('tone').value;
const niche = document.getElementById('niche').value;
const length = document.getElementById('length').value;
const language = document.getElementById('language').value;
const hashtags = document.getElementById('hashtags').checked;
setLoading(true);
hideError();
document.getElementById('results').style.display = 'none';
const prompt = `You are an expert Instagram content creator. Generate
exactly 3 distinct Instagram captions for the following post.
Post description: ${desc}
Tone: ${tone}
Niche: ${niche}
Length: ${length}
Language: ${language}
${hashtags ? 'Include 5-8 relevant hashtags at the end of each caption.'
: 'Do NOT include any hashtags.'}
Return ONLY a JSON array with exactly 3 objects. No explanation, no
markdown, just raw JSON.
Format: [{"vibe":"<one word label>","caption":"<full caption text>"},...]
Make each caption feel genuinely different — vary the hook, structure,
and emotional angle.`;
try {
const res = await fetch('https://openrouter.ai/api/v1/chat/completions',
{
method: 'POST',
headers: {
'Content-Type': 'application/json',
'Authorization': `Bearer ${API_KEY}`,
'HTTP-Referer': window.location.href,
'X-Title': 'CaptionCraft'
},
body: JSON.stringify({
model: 'mistralai/mistral-7b-instruct:free',
max_tokens: 1000,
messages: [{ role: 'user', content: prompt }]
})
});
if (!res.ok) {
const err = await res.json();
throw new Error(err.error?.message || 'API error');
}
const data = await res.json();
const raw = data.choices[0].message.content.replace(/```json|```/g,
'').trim();
const captions = JSON.parse(raw);
renderResults(captions);
} catch (e) {
console.error('API error:', e);
// Show demo captions so users always get value, show error too
renderDemo();
showError('Live AI unavailable right now. Showing sample captions. Error:
' + (e.message || 'unknown'));
} finally {
setLoading(false);
}
}

function renderResults(captions) {
const vibeClasses = ['', 'v2', 'v3'];
const container = document.getElementById('results');
container.innerHTML = '';
captions.forEach((c, i) => {
const div = document.createElement('div');
div.className = 'result-card';
div.style.animationDelay = (i * 0.08) + 's';
div.innerHTML = `
<div class="result-header">
<span class="result-vibe ${vibeClasses[i]}">${c.vibe}</span>
<button class="copy-btn" onclick="copyCaption(this, ${i})">
<svg width="12" height="12" viewBox="0 0 24 24" fill="none"
stroke="currentColor" stroke-width="2"><rect x="9" y="9" width="13"
height="13" rx="2"/><path d="M5 15H4a2 2 0 01-2-2V4a2 2 0 012-2h9a2 2 0
012 2v1"/></svg>
Copy
</button>
</div>
<div class="caption-text" id="caption-${i}">${escHtml(c.caption)}</div>
`;
container.appendChild(div);
});
container.style.display = 'block';
}
function copyCaption(btn, i) {
const text = document.getElementById('caption-' + i).innerText;
navigator.clipboard.writeText(text).then(() => {
btn.classList.add('copied');
btn.innerHTML = `<svg width="12" height="12" viewBox="0 0 24 24"
fill="none" stroke="currentColor" stroke-width="2"><polyline points="20 6
9 17 4 12"/></svg> Copied!`;
setTimeout(() => {
btn.classList.remove('copied');
btn.innerHTML = `<svg width="12" height="12" viewBox="0 0 24 24"
fill="none" stroke="currentColor" stroke-width="2"><rect x="9" y="9"
width="13" height="13" rx="2"/><path d="M5 15H4a2 2 0 01-2-2V4a2 2 0 012-
2h9a2 2 0 012 2v1"/></svg> Copy`;
}, 2000);
});
}
function escHtml(t) { return
t.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }
function setLoading(on) {
document.getElementById('genBtn').disabled = on;
document.getElementById('spinner').style.display = on ? 'block' : 'none';
document.getElementById('btnText').textContent = on ? 'Generating...' :
'Generate 3 Captions';
}
function showError(msg) { const e = document.getElementById('errorMsg');
e.textContent = msg; e.style.display = 'block'; }
